anthropic>=1.2.0
# No speech engine here on purpose. FAM's production voice is Chatterbox,
# which needs torch and is installed separately by
# `pip install -r requirements-chatterbox.txt` - gigabytes, and only useful on
# a machine with a GPU. A laptop, a CI runner and the preview build must all be
# able to install FAM and run its tests without a deep-learning stack; there,
# `/api/health` reports `interim: true` and a placeholder tone plays.
#
# Piper used to sit here as an interim voice and is gone. A second engine that
# can speak is a second engine that can be selected, and an app that quietly
# sounds worse than intended is the failure this project has lost the most time
# to. A tone cannot be mistaken for FAM.
fastapi>=0.110
uvicorn[standard]>=0.27
# app.py imports BaseModel and Field directly. fastapi happens to pull pydantic
# in, so this has always been present by luck rather than by declaration - and
# a dependency that is only there by luck is one nobody notices losing.
pydantic>=2.0

# Not part of the deep-learning stack, and needed without it. `pcm_from_float`
# converts a float waveform to 16-bit PCM on the synthesis path, and the engine
# contract tests exercise that path with chatterbox and torch stubbed out - so
# the one environment that has neither still needs this. It was undeclared, and
# CI caught it the only way an undeclared dependency is ever caught: six tests
# failing with ModuleNotFoundError on a machine that did not happen to have it.
#
# Not replaceable with the stdlib here: a 3-minute episode at 24 kHz is ~4.3M
# samples, and converting those one at a time in Python would put seconds on
# the synthesis path.
numpy>=1.24

# Development only
pytest>=8.0

# No longer development-only. `remote_voice.py` uses this to reach a Chatterbox
# worker on a GPU somewhere else, so with VOICE_BACKEND=remote it is on the
# synthesis path of every episode - a CPU deployment that skipped it would
# install cleanly and then be unable to speak. It is also what
# fastapi.testclient runs on.
#
# The Anthropic SDK uses httpx2, which it pulls in itself - do not add httpx
# expecting the SDK to use it.
httpx>=0.27
